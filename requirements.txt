# Python dependencies of the Safety Island flow. They are installed in the
# project virtual environment (`.venv`) by `setup_env.py`, which `bootstrap.sh`
# runs for you.
#
# Every package (including the transitive MarkupSafe) is pinned to an exact
# version so that the environment is identical on every machine. The pins have
# wheels for Python 3.9 up to 3.12.

# ---- Register generator (`make safed-hw-gen`) ------------------------------
# `regtool.py` is shipped in the `register_interface` Bender dependency and is
# derived from OpenTitan's `reggen`. It is not needed to simulate, the generated
# files are committed.

# Reads the register description (`rtl/soc_ctrl/safety_soc_ctrl_regs.hjson`)
hjson==3.1.0

# Template engine that renders the generated RTL (`*_reg_pkg.sv`, `*_reg_top.sv`)
Mako==1.3.12
MarkupSafe==2.1.5

# Imported at module level by reggen (generators for DV / FPV collateral)
PyYAML==6.0.3

# Imported at module level by reggen (markdown documentation helpers)
tabulate==0.9.0

# reggen imports `pkg_resources` to locate its templates. Python >= 3.12 venvs
# do not ship setuptools, and recent setuptools releases no longer contain
# `pkg_resources`, so it has to be installed explicitly and kept on an old
# release. 75.3.4 is the last release that supports Python 3.8.
setuptools==75.3.4

# ---- Simulation (`make run` in `sw/tests/<test>`) ---------------------------
# Used by the scripts of `pulp-runtime` (`bin/stim_utils.py`,
# `bin/plp_flash_stimuli.py`) and by `boot/stim_utils.py` to read the ELF.
pyelftools==0.32

# Used by `pulp-runtime/bin/slm_hyper.py`. 1.26.4 is the last release that
# supports Python 3.9 (and it supports up to 3.12).
numpy==1.26.4
